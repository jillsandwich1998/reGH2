# gh2k_sync: IRQL, spin locks, critical sections and TLS slots on guest memory (Phase 2.7, K.2). Plain C++17: no SDK, no guest macros.
find_package(Threads REQUIRED)
add_library(gh2k_sync STATIC sync.cpp objects.cpp)
target_include_directories(gh2k_sync PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(gh2k_sync PUBLIC Threads::Threads)
target_compile_options(gh2k_sync PRIVATE -Wall -Wextra)
set_target_properties(gh2k_sync PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(gh2_sync_selftest EXCLUDE_FROM_ALL sync_selftest.cpp)
target_link_libraries(gh2_sync_selftest PRIVATE gh2k_sync)
# the same test with ThreadSanitizer (its own copy of the library)
add_library(gh2k_sync_tsan STATIC EXCLUDE_FROM_ALL sync.cpp objects.cpp)
target_include_directories(gh2k_sync_tsan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_options(gh2k_sync_tsan PRIVATE -fsanitize=thread -g)
add_executable(gh2_sync_selftest_tsan EXCLUDE_FROM_ALL sync_selftest.cpp)
target_link_libraries(gh2_sync_selftest_tsan PRIVATE gh2k_sync_tsan Threads::Threads -fsanitize=thread)
target_compile_options(gh2_sync_selftest_tsan PRIVATE -fsanitize=thread -g)
