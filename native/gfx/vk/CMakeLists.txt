# gh2vk: the Vulkan device, memory and swapchain of the native renderer (Phase 3). Plain C++17 + the system Vulkan loader. Vulkan Memory Allocator (a single MIT-licensed header) comes
# from the SDK's include directory for now and is exposed through a directory that holds only that header (to be vendored into third_party).
# Vulkan: the system's headers and loader (a developer's machine), else the SDK's headers and the loader library the AppImage carries (the player's build: no -dev packages needed;
# the installer's launcher exports GH2_VULKAN_LOADER=<path of libvulkan.so.1>). At run time the loader is the host's libvulkan.so.1 (found by its soname).
# The AppImage's build (GH2_VULKAN_LOADER set) always links the loader the AppImage carries: the host's libvulkan may need a newer glibc than the toolchain's sysroot has.
if(NOT EXISTS "$ENV{GH2_VULKAN_LOADER}")
    find_package(Vulkan QUIET)
endif()
if(NOT TARGET Vulkan::Vulkan)
    find_path(GH2_VK_HEADERS vulkan/vulkan.h HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include NO_DEFAULT_PATH)
    if(EXISTS "$ENV{GH2_VULKAN_LOADER}")
        set(GH2_VK_LOADER "$ENV{GH2_VULKAN_LOADER}" CACHE FILEPATH "Vulkan loader linked against" FORCE)
    else()
        find_library(GH2_VK_LOADER NAMES vulkan libvulkan.so.1)
    endif()
    if(NOT GH2_VK_HEADERS OR NOT GH2_VK_LOADER)
        message(FATAL_ERROR "Vulkan headers (SDK include/vulkan) or the loader (libvulkan.so.1, GH2_VULKAN_LOADER) not found")
    endif()
    add_library(Vulkan::Vulkan INTERFACE IMPORTED)
    set_target_properties(Vulkan::Vulkan PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${GH2_VK_HEADERS}" INTERFACE_LINK_LIBRARIES "${GH2_VK_LOADER}")
endif()
find_path(GH2_VMA_SDK_INCLUDE vk_mem_alloc.h HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES include NO_DEFAULT_PATH)
if(NOT GH2_VMA_SDK_INCLUDE)
    message(FATAL_ERROR "vk_mem_alloc.h not found next to the SDK")
endif()
set(GH2_VMA_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/vma_only")
file(MAKE_DIRECTORY "${GH2_VMA_INCLUDE_DIR}")
file(CREATE_LINK "${GH2_VMA_SDK_INCLUDE}/vk_mem_alloc.h" "${GH2_VMA_INCLUDE_DIR}/vk_mem_alloc.h" SYMBOLIC)
# vulkan.h with VK_USE_PLATFORM_XLIB_KHR includes <X11/Xlib.h>: without the X11 development files a three-typedef stand-in is used (see x11_shim/X11/Xlib.h)
if(CMAKE_SYSROOT)   # the AppImage's hermetic toolchain only sees its own sysroot, whatever the host has in /usr/include
    find_path(GH2_X11_INCLUDE X11/Xlib.h HINTS "${CMAKE_SYSROOT}/usr/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
else()
    find_path(GH2_X11_INCLUDE X11/Xlib.h)
endif()
add_library(gh2vk STATIC context.cpp vma_impl.cpp)
if(NOT GH2_X11_INCLUDE)
    target_include_directories(gh2vk SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/x11_shim")
endif()
target_include_directories(gh2vk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(gh2vk PRIVATE SYSTEM "${GH2_VMA_INCLUDE_DIR}")
target_compile_definitions(gh2vk PUBLIC VK_USE_PLATFORM_WAYLAND_KHR VK_USE_PLATFORM_XLIB_KHR)
target_link_libraries(gh2vk PUBLIC Vulkan::Vulkan)
target_compile_options(gh2vk PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
set_target_properties(gh2vk PROPERTIES POSITION_INDEPENDENT_CODE ON)
set_source_files_properties(vma_impl.cpp PROPERTIES COMPILE_OPTIONS "-w")
